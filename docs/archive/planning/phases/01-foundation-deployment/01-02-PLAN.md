Create `app/[locale]/layout.tsx` combining Once UI's CSS-import-order requirement with next-intl's locale validation + static rendering (synthesized from RESEARCH.md Pattern 2 + Context7 `/amannn/next-intl` since no single source gives the combined shape):

```tsx
import '@once-ui-system/core/css/styles.css';
import '@once-ui-system/core/css/tokens.css';
import '@/resources/custom.css';
import classNames from 'classnames';
import { notFound } from 'next/navigation';
import { hasLocale, NextIntlClientProvider } from 'next-intl';
import { setRequestLocale } from 'next-intl/server';
import { routing } from '@/i18n/routing';
import { fonts } from '@/resources/once-ui.config';
import { Providers } from '@/components/Providers';

export function generateStaticParams() {
  return routing.locales.map((locale) => ({ locale }));
}

export default async function RootLayout({
  children,
  params,
}: {
  children: React.ReactNode;
  params: Promise<{ locale: string }>;
}) {
  const { locale } = await params;
  if (!hasLocale(routing.locales, locale)) {
    notFound();
  }
  setRequestLocale(locale);

  return (
    <html
      lang={locale}
      className={classNames(
        fonts.heading.variable,
        fonts.body.variable,
        fonts.label.variable,
        fonts.code.variable,
      )}
    >
      <body>
        <NextIntlClientProvider>
          <Providers>{children}</Providers>
        </NextIntlClientProvider>
      </body>
    </html>
  );
}
```

CSS import order is load-bearing per RESEARCH.md Pattern 2 — do not reorder: `@once-ui-system/core/css/styles.css` → `@once-ui-system/core/css/tokens.css` → `@/resources/custom.css`, before any component import.
